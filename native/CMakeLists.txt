cmake_minimum_required(VERSION 3.20)
project(VideoManagerNative LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The initial platform adapter targets Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FFMPEG_ROOT "" CACHE PATH "Extracted LGPL shared FFmpeg SDK")
add_executable(vm-fs filesystem/main.cpp)
target_link_libraries(vm-fs PRIVATE bcrypt)
add_executable(vm-media media/main.cpp)
target_include_directories(vm-media PRIVATE "${FFMPEG_ROOT}/include")
target_link_directories(vm-media PRIVATE "${FFMPEG_ROOT}/lib")
target_link_libraries(vm-media PRIVATE avformat avcodec avutil swscale)
if(MINGW)
  target_link_options(vm-fs PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(vm-media PRIVATE -static-libgcc -static-libstdc++)
endif()
install(TARGETS vm-fs vm-media RUNTIME DESTINATION .)
